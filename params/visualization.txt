# =====================================================================
# visualization.txt  --  turn on the interactive 3D viewer (Qt + OpenGL)
# Builds on: scoring.txt (i.e. the complete simulation)
# =====================================================================
includeFile = scoring.txt

b:Gr/Enable  = "True"
b:Ts/UseQt   = "True"          # interactive Qt window with menus and a command line
s:Gr/ViewA/Type   = "OpenGL"
i:Gr/ViewA/WindowSizeX = 1000
i:Gr/ViewA/WindowSizeY = 800
d:Gr/ViewA/Theta  = 90 deg     # look from the side (+Y axis) ...
d:Gr/ViewA/Phi    = 0 deg
s:Gr/ViewA/Projection = "Perspective"
d:Gr/ViewA/PerspectiveAngle = 30 deg
u:Gr/ViewA/Zoom   = 1.5
b:Gr/ViewA/IncludeAxes = "True"   # red = X, green = Y, blue = Z
d:Gr/ViewA/AxesSize    = 5 cm
b:Gr/ViewA/HiddenLineRemovalForTrajectories = "True"

# Colour tracks by particle type
s:Gr/ViewA/ColorBy = "ParticleType"
sv:Gr/ViewA/ColorByParticleTypeNames  = 4 "proton" "e-" "gamma" "neutron"
sv:Gr/ViewA/ColorByParticleTypeColors = 4 "red" "yellow" "green" "grey"

# Graphics is slow: one thread and only a handful of protons.
i:Ts/NumberOfThreads = 1
i:Ts/ShowHistoryCountAtInterval = 10
b:Ts/PauseBeforeQuit = "True"     # keep the window open at the end
